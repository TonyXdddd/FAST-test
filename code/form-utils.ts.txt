//>fu-types
/** Нативные input этих типов блокируют неявную отправку (HTML §4.10.22.2) */
const submitBlockingElementTypes = new Set([
  'text', 'search', 'url', 'tel', 'email', 'password', 'date', 'month', 'week', 'time', 'datetime-local', 'number',
]);
//<fu-types

//>fu-custom-types
/** Уникальные типы наших компонентов, которые тоже считаются текстовыми полями */
const customSubmitBlockingElementTypes = new Set([
  'date-interval', 'time-interval', 'year',
]);
//<fu-custom-types

//>fu-custom
/** Кастомный элемент: в имени тега всегда есть дефис */
function isCustomElement($element: Element): boolean {
  return $element.localName.includes('-');
}

/** Свойство type компонента, если это строка */
function getElementType($element: Element): string | undefined {
  const type = ($element as Element & { type?: unknown }).type;
  return typeof type === 'string' ? type : undefined;
}
//<fu-custom

//>fu-isSubmitButton
function isSubmitButton($element: Element): boolean {
  if ($element instanceof HTMLButtonElement) return $element.type === 'submit';
  if ($element instanceof HTMLInputElement) return $element.type === 'submit' || $element.type === 'image';
//>fu-custom-button
  return isCustomElement($element) && getElementType($element) === 'submit';
//<fu-custom-button
}
//<fu-isSubmitButton

//>fu-isSubmitBlocking
function isSubmitBlocking($element: Element): boolean {
  if ($element instanceof HTMLInputElement) {
    return submitBlockingElementTypes.has($element.type);
  }
  if (!isCustomElement($element)) {
    return false; // textarea, select, button, fieldset, output
  }
//>fu-custom-check
  const type = getElementType($element);
  return type !== undefined
    && (submitBlockingElementTypes.has(type) || customSubmitBlockingElementTypes.has(type));
//<fu-custom-check
}
//<fu-isSubmitBlocking

//>fu-getDefaultButton
/** Первая submit-кнопка формы в порядке документа, включая кнопки с form="id" вне <form> */
function getDefaultButton($form: HTMLFormElement): HTMLElement | null {
  let button = (Array.from($form.elements).find(isSubmitButton) as HTMLElement | undefined) ?? null;

//>fu-image
  // <input type="image"> не входит в form.elements
  const root = $form.getRootNode() as Document | ShadowRoot;
  const image = Array.from(root.querySelectorAll('input')).find($ => $.type === 'image' && $.form === $form);
  if (image && (!button || button.compareDocumentPosition(image) & Node.DOCUMENT_POSITION_PRECEDING)) {
    button = image;
  }
//<fu-image

  return button;
}
//<fu-getDefaultButton

//>fu-mimic
/**
 * Повторяет неявную отправку формы по Enter (HTML Standard §4.10.22.2).
 * @returns true — Enter обработан: нашлась кнопка по умолчанию или вызван requestSubmit()
 */
export function mimicFormSubmitBehavior($elementInternals: ElementInternals): boolean {
//>fu-mimic-form
  const form = $elementInternals.form;
  if (!form) {
    return false;
  }
//<fu-mimic-form

//>fu-mimic-button
  const button = getDefaultButton(form);
  if (button) {
//>fu-mimic-disabled
    // disabled (в т.ч. через <fieldset disabled>) — ничего не делать и следующую кнопку не искать
    if (!button.matches(':disabled')) {
      button.click();
    }
//<fu-mimic-disabled
    return true;
  }
//<fu-mimic-button

//>fu-mimic-count
  if (Array.from(form.elements).filter(isSubmitBlocking).length < 2) {
    form.requestSubmit();
    return true;
  }
//<fu-mimic-count

//>fu-mimic-false
  return false;
//<fu-mimic-false
}
//<fu-mimic
